𑆃𑆡𑆳𑆲𑆫𑆼𑆠𑆴 𑆧𑇀𑆫𑆷𑆪𑆳𑆠𑇀- 𑆯𑆳𑆫𑆢𑆳𑆤𑇀𑆤𑆮𑆳𑆁𑆱𑇀𑆠𑆴𑆬𑆳𑆤𑇀 𑆱𑆩𑇀𑆥𑆢𑆶𑆥𑆼𑆠𑆳𑆤𑇀; 𑆠𑆳𑆤𑆳𑆲𑆸𑆠𑇀𑆪 𑆱𑆶𑆤𑆴𑆰𑇀𑆥𑆷𑆠𑆳𑆤𑇀𑆤𑆴𑆰𑇀𑆥𑆷𑆪, 𑆱𑆶𑆯𑆶𑆢𑇀𑆣𑆳𑆤𑇀 𑆯𑆾𑆣𑆪𑆴𑆠𑇀𑆮𑆳 , 𑆮𑆴𑆝𑆕𑇀𑆓𑆑𑆰𑆳𑆪𑆼 𑆱𑆶𑆒𑆾𑆰𑇀𑆟𑆼 𑆥𑇀𑆫𑆑𑇀𑆰𑆴𑆥𑇀𑆪 𑆤𑆴𑆫𑇀𑆮𑆳𑆥𑆪𑆼𑆢𑆳𑆢𑆾𑆰𑆓𑆩𑆤𑆳𑆠𑇀, 𑆓𑆠𑆢𑆾𑆰𑆳𑆤𑆨𑆴𑆱𑆩𑆵𑆑𑇀𑆰𑇀𑆪, 𑆱𑆶𑆥𑇀𑆫𑆬𑆷𑆤𑆳𑆤𑇀 𑆥𑇀𑆫𑆬𑆶𑆚𑇀𑆖𑇀𑆪, 𑆥𑆶𑆤𑆫𑆼𑆮 𑆱𑆶𑆤𑆴𑆰𑇀𑆥𑆷𑆠𑆳𑆤𑇀 𑆤𑆴𑆰𑇀𑆥𑆷𑆪, 𑆱𑆶𑆯𑆶𑆢𑇀𑆣𑆳𑆤𑇀 𑆯𑆾𑆣𑆪𑆴𑆠𑇀𑆮𑆳, 𑆮𑆴𑆝𑆕𑇀𑆓𑆑𑆰𑆳𑆪𑆼𑆟 𑆠𑇀𑆫𑆴𑆂𑆱𑆥𑇀𑆠𑆑𑆸𑆠𑇀𑆮𑆂 𑆱𑆶𑆥𑆫𑆴𑆨𑆳𑆮𑆴𑆠𑆳𑆤𑇀 𑆨𑆳𑆮𑆪𑆴𑆠𑇀𑆮𑆳, 𑆄𑆠𑆥𑆼 𑆯𑆾𑆰𑆪𑆴𑆠𑇀𑆮𑆳, 𑆇𑆬𑆷(𑆢𑆷)𑆒𑆬𑆼 𑆱𑆕𑇀𑆑𑇀𑆰𑆶𑆢𑇀𑆪, 𑆢𑆸𑆰𑆢𑆴 𑆥𑆶𑆤𑆂 𑆯𑇀𑆬𑆑𑇀𑆰𑇀𑆟𑆥𑆴𑆰𑇀𑆛𑆳𑆤𑇀 𑆑𑆳𑆫𑆪𑆴𑆠𑇀𑆮𑆳, 𑆢𑇀𑆫𑆾𑆟𑇀𑆪𑆳𑆩𑆨𑇀𑆪𑆮𑆣𑆳𑆪, 𑆮𑆴𑆝𑆕𑇀𑆓𑆑𑆰𑆳𑆪𑆼𑆟 𑆩𑆶𑆲𑆶𑆫𑇀𑆩𑆶𑆲𑆶𑆫𑆮𑆱𑆴𑆚𑇀𑆖𑆤𑇀 𑆥𑆳𑆟𑆴𑆩𑆫𑇀𑆢𑆩𑆼𑆮 𑆩𑆫𑇀𑆢𑆪𑆼𑆠𑇀; 𑆠𑆱𑇀𑆩𑆴𑆁𑆱𑇀𑆠𑆶 𑆒𑆬𑆶 𑆥𑇀𑆫𑆥𑆵𑆝𑇀𑆪𑆩𑆳𑆤𑆼 𑆪𑆠𑇀𑆠𑆽𑆬𑆩𑆶𑆢𑆴𑆪𑆳𑆠𑇀𑆠𑆠𑇀 𑆥𑆳𑆟𑆴𑆨𑇀𑆪𑆳𑆁 𑆥𑆫𑇀𑆪𑆳𑆢𑆳𑆪, 𑆯𑆶𑆖𑆿 𑆢𑆸𑆞𑆼 𑆑𑆬𑆯𑆼 𑆤𑇀𑆪𑆱𑇀𑆪𑆳𑆤𑆶𑆓𑆶𑆥𑇀𑆠𑆁 𑆤𑆴𑆣𑆳𑆥𑆪𑆼𑆠𑇀𑇆𑇒𑇕𑇆
𑆃𑆡𑆳𑆲𑆫𑆼𑆠𑆴 𑆧𑇀𑆫𑆷𑆪𑆳𑆠𑇀- 𑆠𑆴𑆬𑇀𑆮𑆑𑆾𑆢𑇀𑆢𑆳𑆬𑆑𑆪𑆾𑆫𑇀𑆢𑇀𑆮𑆿 𑆧𑆴𑆬𑇀𑆮𑆩𑆳𑆠𑇀𑆫𑆿 𑆥𑆴𑆟𑇀𑆝𑆿 𑆯𑇀𑆬𑆑𑇀𑆰𑇀𑆟𑆥𑆴𑆰𑇀𑆛𑆿 𑆮𑆴𑆝𑆕𑇀𑆓𑆑𑆰𑆳𑆪𑆼𑆟, 𑆠𑆢𑆫𑇀𑆣𑆩𑆳𑆠𑇀𑆫𑆿 𑆯𑇀𑆪𑆳𑆩𑆳𑆠𑇀𑆫𑆴𑆮𑆸𑆠𑆪𑆾𑆂, 𑆃𑆠𑆾𑇁𑆫𑇀𑆣𑆩𑆳𑆠𑇀𑆫𑆿 𑆢𑆤𑇀𑆠𑆵𑆢𑇀𑆫𑆮𑆤𑇀𑆠𑇀𑆪𑆾𑆂, 𑆃𑆠𑆾𑇁𑆫𑇀𑆣𑆩𑆳𑆠𑇀𑆫𑆿 𑆖 𑆖𑆮𑇀𑆪𑆖𑆴𑆠𑇀𑆫𑆑𑆪𑆾𑆫𑆴𑆠𑆴𑇅 𑆍𑆠𑆁 𑆱𑆩𑇀𑆨𑆳𑆫𑆁 𑆮𑆴𑆝𑆕𑇀𑆓𑆑𑆰𑆳𑆪𑆱𑇀𑆪𑆳𑆫𑇀𑆣𑆳𑆞𑆑𑆩𑆳𑆠𑇀𑆫𑆼𑆟 𑆥𑇀𑆫𑆠𑆴𑆱𑆁𑆱𑆸𑆘𑇀𑆪, 𑆠𑆠𑇀𑆠𑆽𑆬𑆥𑇀𑆫𑆱𑇀𑆡𑆁 𑆱𑆩𑆳𑆮𑆳𑆥𑇀𑆪, 𑆱𑆫𑇀𑆮𑆩𑆳𑆬𑆾𑆝𑇀𑆪, 𑆩𑆲𑆠𑆴 𑆥𑆫𑇀𑆪𑆾𑆓𑆼 𑆱𑆩𑆳𑆱𑆴𑆖𑇀𑆪𑆳𑆓𑇀𑆤𑆳𑆮𑆣𑆴𑆯𑇀𑆫𑆴𑆠𑇀𑆪𑆳𑆱𑆤𑆼 𑆱𑆶𑆒𑆾𑆥𑆮𑆴𑆰𑇀𑆛𑆂 𑆱𑆫𑇀𑆮𑆠𑆂 𑆱𑇀𑆤𑆼𑆲𑆩𑆮𑆬𑆾𑆑𑆪𑆤𑇀𑆤𑆘𑆱𑇀𑆫𑆁 𑆩𑆸𑆢𑇀𑆮𑆓𑇀𑆤𑆴𑆤𑆳 𑆱𑆳𑆣𑆪𑆼𑆢𑇀𑆢𑆫𑇀𑆮𑇀𑆪𑆳 𑆱𑆠𑆠𑆩𑆮𑆔𑆛𑇀𑆛𑆪𑆤𑇀𑇅 𑆱 𑆪𑆢𑆳 𑆘𑆳𑆤𑆵𑆪𑆳𑆢𑇀𑆮𑆴𑆫𑆩𑆠𑆴 𑆯𑆧𑇀𑆢𑆂, 𑆥𑇀𑆫𑆯𑆳𑆩𑇀𑆪𑆠𑆴 𑆖 𑆦𑆼𑆤𑆂, 𑆥𑇀𑆫𑆱𑆳𑆢𑆩𑆳𑆥𑆢𑇀𑆪𑆠𑆼 𑆱𑇀𑆤𑆼𑆲𑆂, 𑆪𑆡𑆳𑆱𑇀𑆮𑆁 𑆖 𑆓𑆤𑇀𑆣𑆮𑆫𑇀𑆟𑆫𑆱𑆾𑆠𑇀𑆥𑆠𑇀𑆠𑆴𑆂, 𑆱𑆁𑆮𑆫𑇀𑆠𑆠𑆼 𑆖 𑆨𑆽𑆰𑆘𑇀𑆪𑆩𑆕𑇀𑆓𑆶𑆬𑆴𑆨𑇀𑆪𑆳𑆁 𑆩𑆸𑆢𑇀𑆪𑆩𑆳𑆤𑆩𑆤𑆠𑆴𑆩𑆸𑆢𑇀𑆮𑆤𑆠𑆴𑆢𑆳𑆫𑆶𑆟𑆩𑆤𑆕𑇀𑆓𑆶𑆬𑆴𑆓𑇀𑆫𑆳𑆲𑆴 𑆖𑆼𑆠𑆴, 𑆱 𑆑𑆳𑆬𑆱𑇀𑆠𑆱𑇀𑆪𑆳𑆮𑆠𑆳𑆫𑆟𑆳𑆪𑇅 𑆠𑆠𑆱𑇀𑆠𑆩𑆮𑆠𑆳𑆫𑇀𑆪 𑆯𑆵𑆠𑆵𑆨𑆷𑆠𑆩𑆲𑆠𑆼𑆤 𑆮𑆳𑆱𑆱𑆳 𑆥𑆫𑆴𑆥𑆷𑆪, 𑆯𑆶𑆖𑆿 𑆢𑆸𑆞𑆼 𑆑𑆬𑆯𑆼 𑆱𑆩𑆳𑆱𑆴𑆖𑇀𑆪, 𑆥𑆴𑆣𑆳𑆤𑆼𑆤 𑆥𑆴𑆣𑆳𑆪, 𑆯𑆶𑆑𑇀𑆬𑆼𑆤 𑆮𑆱𑇀𑆠𑇀𑆫𑆥𑆛𑇀𑆛𑆼𑆤𑆳𑆮𑆖𑇀𑆗𑆳𑆢𑇀𑆪, 𑆱𑆷𑆠𑇀𑆫𑆼𑆟 𑆱𑆶𑆧𑆢𑇀𑆣𑆁 𑆱𑆶𑆤𑆴𑆓𑆶𑆥𑇀𑆠𑆁 𑆤𑆴𑆣𑆳𑆥𑆪𑆼𑆠𑇀𑇅 𑆠𑆠𑆾𑇁𑆱𑇀𑆩𑆽 𑆩𑆳𑆠𑇀𑆫𑆳𑆁 𑆥𑇀𑆫𑆪𑆖𑇀𑆗𑆼𑆠𑇀 𑆥𑆳𑆤𑆳𑆪, 𑆠𑆼𑆤 𑆱𑆳𑆣𑆶 𑆮𑆴𑆫𑆴𑆖𑇀𑆪𑆠𑆼; 𑆱𑆩𑇀𑆪𑆓𑆥𑆲𑆸𑆠𑆢𑆾𑆰𑆱𑇀𑆪 𑆖𑆳𑆤𑆶𑆥𑆷𑆫𑇀𑆮𑆵 𑆪𑆡𑆾𑆑𑇀𑆠𑆳𑇅 𑆠𑆠𑆯𑇀𑆖𑆽𑆤𑆩𑆤𑆶𑆮𑆳𑆱𑆪𑆼𑆢𑆤𑆶𑆮𑆳𑆱𑆤𑆑𑆳𑆬𑆼𑇅 𑆍𑆠𑆼𑆤𑆽𑆮 𑆖 𑆥𑆳𑆑𑆮𑆴𑆣𑆴𑆤𑆳 𑆱𑆫𑇀𑆰𑆥𑆳𑆠𑆱𑆵𑆑𑆫𑆚𑇀𑆘𑆑𑆾𑆰𑆳𑆠𑆑𑆵𑆱𑇀𑆤𑆼𑆲𑆳𑆤𑆶𑆥𑆑𑆬𑇀𑆥𑇀𑆪 𑆥𑆳𑆪𑆪𑆼𑆠𑇀 𑆱𑆫𑇀𑆮𑆮𑆴𑆯𑆼𑆰𑆳𑆤𑆮𑆼𑆑𑇀𑆰𑆩𑆳𑆟𑆂𑇅 𑆠𑆼𑆤𑆳𑆓𑆢𑆾 𑆨𑆮𑆠𑆴𑇆𑇒𑇖𑇆
